  <script>
    // ฟังก์ชันเปิด-ปิด Sidebar สำหรับมือถือ
    function toggleSidebar() {
      const sidebar = document.getElementById('sidebar');
      sidebar.classList.toggle('show');
    }

    // ฟังก์ชันโหลดหน้าเว็บ (SPA)
    function loadPage(filename, element) {
      // 1. ปิด Sidebar ทันที (กรณีมือถือ)
      document.getElementById('sidebar').classList.remove('show');

      // 2. แสดงสถานะกำลังโหลด
      const area = document.getElementById('display-area');
      const loader = document.getElementById('loader');
      
      area.style.opacity = '0.2';
      loader.style.display = 'block';

      // 3. ปรับสถานะปุ่มเมนู (Active State)
      if(element) {
        document.querySelectorAll('.nav-link').forEach(link => link.classList.remove('active'));
        element.classList.add('active');
      }

      // 4. เรียกใช้ Google Apps Script เพื่อดึงไฟล์ HTML
      google.script.run
        .withSuccessHandler(function(htmlContent) {
          area.innerHTML = htmlContent;
          area.style.opacity = '1';
          loader.style.display = 'none';
          
          // เลื่อนขึ้นด้านบนสุดทุกครั้งที่เปลี่ยนหน้า
          window.scrollTo({ top: 0, behavior: 'smooth' });
        })
        .withFailureHandler(function(err) {
          area.innerHTML = '<div class="alert alert-danger">เกิดข้อผิดพลาดในการโหลดข้อมูล: ' + err + '</div>';
          loader.style.display = 'none';
          area.style.opacity = '1';
        })
        .loadPartialView(filename);
    }

    // เมื่อเปิดหน้าเว็บครั้งแรก ให้โหลดหน้า Profile ทันที
    window.onload = () => {
      loadPage('Profile', document.querySelector('.nav-link.active'));
    };
  </script>
