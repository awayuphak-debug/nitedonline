// ฟังก์ชันสำหรับดึงข้อมูลมาสร้างกราฟ
function getSummaryData() {
  const ssId = '1rZd4MdvfLHam5tH2r9BWSy8BGsydc9uWoamsmw4DJ_I'; // ID ของ Google Sheet
  const sheet = SpreadsheetApp.openById(ssId).getSheets()[0];
  const data = sheet.getDataRange().getValues();
  
  // ข้อมูลจาก Sheet1 ประกอบด้วย: ที่, รหัสโรงเรียน, โรงเรียน, จำนวนนักเรียน, อำเภอ 
  const districtMap = {};
  
  // เริ่มวนลูปที่แถวที่ 1 (ข้ามหัวตาราง)
  for (let i = 1; i < data.length; i++) {
    const studentCount = parseInt(data[i][3]); // คอลัมน์ "จำนวนนักเรียน"
    const district = data[i][4];               // คอลัมน์ "อำเภอ"
    
    if (district) {
      districtMap[district] = (districtMap[district] || 0) + studentCount;
    }
  }
  
  // แปลง Object เป็น Array สำหรับใช้งานในกราฟ
  const labels = Object.keys(districtMap);
  const values = Object.values(districtMap);
  
  return { labels: labels, values: values };
}

                  minRotation: 45,

        });
